local function createTab(name, active)
    local b = Instance.new("TextButton", tabHolder)
    b.Size = UDim2.new(1, -10, 0, 35)
    b.Position = UDim2.new(0, 5, 0, (#tabHolder:GetChildren()-1) * 40 + 10)
    b.Text = name
    b.Font = Enum.Font.GothamBold
    b.TextSize = 11
    b.BackgroundColor3 = active and Color3.fromRGB(50, 50, 50) or Color3.fromRGB(20, 20, 20)
    b.TextColor3 = Color3.new(1,1,1)
    Instance.new("UICorner", b)
    
    local p = Instance.new("ScrollingFrame", container)
    p.Size = UDim2.new(1, 0, 1, 0)
    p.BackgroundTransparency = 1
    p.Visible = active
    p.CanvasSize = UDim2.new(0,0,0,320)
    p.ScrollBarThickness = 0
    local layout = Instance.new("UIListLayout", p)
    layout.Padding = UDim.new(0, 5)

    b.Activated:Connect(function()
        for _, v in pairs(tabs) do v.btn.BackgroundColor3 = Color3.fromRGB(20, 20, 20); v.page.Visible = false end
        b.BackgroundColor3 = Color3.fromRGB(50, 50, 50)
        p.Visible = true
    end)
    tabs[name] = {btn = b, page = p}
    return p
end

local jumpPower = 240 
local repeatSpeed = 0.01 
local holdingJump = false
local dragInput, dragStart, startPos

local function doJump()
    local char = player.Character
    if char then
        local root = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChild("Humanoid")
        if root and hum and hum.Health > 0 then
            hum:ChangeState(Enum.HumanoidStateType.Jumping)
            root.AssemblyLinearVelocity = Vector3.new(root.AssemblyLinearVelocity.X, jumpPower, root.AssemblyLinearVelocity.Z)
        end
    end
end

jumpButton.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
        holdingJump = true
        task.spawn(function()
            while holdingJump do doJump(); task.wait(repeatSpeed) end
        end)
        dragInput = input; dragStart = input.Position; startPos = jumpButton.Position
    end
end)

UIS.InputChanged:Connect(function(input)
    if input == dragInput and holdingJump then
        local delta = input.Position - dragStart
        jumpButton.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
    end
end)

jumpButton.InputEnded:Connect(function(input)
    if input == dragInput then holdingJump = false; dragInput = nil end
end)

RunService.RenderStepped:Connect(function()
    local char = player.Character
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    if hrp then
        if LockEnabled and CurrentTarget then
            camera.CFrame = camera.CFrame:Lerp(CFrame.new(camera.CFrame.Position, CurrentTarget.Position), SMOOTHNESS)
        end
        if _G.currentSpeed > 16 and char.Humanoid.MoveDirection.Magnitude > 0 then
            hrp.Velocity = Vector3.new(char.Humanoid.MoveDirection.X * _G.currentSpeed, hrp.Velocity.Y, char.Humanoid.MoveDirection.Z * _G.currentSpeed)
        end
        if _G.WalkOnWater then
            waterPart.CFrame = CFrame.new(hrp.Position.X, 0, hrp.Position.Z); waterPart.CanCollide = true
        else waterPart.CanCollide = false end
    end
end)

task.spawn(function()
    while task.wait(0.1) do
        local best, shortest = nil, 3000 -- UPDATED: Changed from 2000 to 3000
        local root = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
        if root then
            local potential = {}
            for _, p in pairs(Players:GetPlayers()) do if p ~= player and p.Character then table.insert(potential, p.Character) end end
            if workspace:FindFirstChild("Enemies") then for _, e in pairs(workspace.Enemies:GetChildren()) do table.insert(potential, e) end end
            for _, char in pairs(potential) do
                local hPart = char:FindFirstChild("HumanoidRootPart")
                if hPart and char:FindFirstChild("Humanoid") and char.Humanoid.Health > 0 then
                    local mag = (hPart.Position - root.Position).Magnitude
                    if mag < shortest then
                        local isNPC = not Players:GetPlayerFromCharacter(char)
                        if (isNPC and (_G.SilentAim_NPC or (LockEnabled and _G.CamLock_NPC))) or (not isNPC and (_G.SilentAim_Player or (LockEnabled and _G.CamLock_Player))) then
                            shortest = mag; best = hPart
                        end
                    end
                end
            end
            CurrentTarget = best
        end
    end
end)

task.spawn(function()
    while task.wait() do
        if _G.FastAttack and player.Character then ReplicatedStorage.Remotes.CommF_:InvokeServer("Attack", {}) end
    end
end)

local old; old = hookmetamethod(game, "__namecall", function(self, ...)
    local method = getnamecallmethod()
    local args = {...}
    if not checkcaller() and (method == "FireServer" or method == "InvokeServer") and CurrentTarget then
        local pred = CurrentTarget.Position + (CurrentTarget.Velocity * PREDICTION_STRENGTH)
        for i, v in pairs(args) do
            if typeof(v) == "Vector3" then args[i] = pred
            elseif typeof(v) == "CFrame" then args[i] = CFrame.new(v.Position, pred) end
        end
    end
    return old(self, unpack(args))
end)

local function drag(gui)
    local d, s, sp; gui.InputBegan:Connect(function(i) if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then d = true; s = i.Position; sp = gui.Position end end)
    gui.InputChanged:Connect(function(i) if d and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then local delta = i.Position - s; gui.Position = UDim2.new(sp.X.Scale, sp.X.Offset + delta.X, sp.Y.Scale, sp.Y.Offset + delta.Y) end end)
    gui.InputEnded:Connect(function() d = false end)
end
