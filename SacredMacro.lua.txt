-- ── Guardar / Cargar ──────────────────────────────────────────
local function saveMacros()
    pcall(function() writefile(SAVE_FILE, HttpService:JSONEncode(macros)) end)
end

local function loadMacros()
    pcall(function()
        if isfile and isfile(SAVE_FILE) then
            local d = HttpService:JSONDecode(readfile(SAVE_FILE))
            if type(d) == "table" then macros = d end
        end
    end)
end

loadMacros()

-- ── Input ─────────────────────────────────────────────────────
local function pressKey(keyStr, holdTime)
    pcall(function()
        local char = lp.Character
        local hrp  = char and char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        VIM:SendKeyEvent(true,  Enum.KeyCode[keyStr], false, hrp)
        task.wait(holdTime or 0.15)
        VIM:SendKeyEvent(false, Enum.KeyCode[keyStr], false, hrp)
    end)
end

local function equipWeapon(wType)
    if wType == "Fruit" then return end
    pcall(function()
        local char = lp.Character
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hum then return end
        for _, tool in ipairs(lp.Backpack:GetChildren()) do
            if tool:IsA("Tool") then
                local tt = (tool.ToolTip or ""):lower()
                if wType == "Melee" and (tt:find("melee") or tt:find("fighting") or tt:find("godhuman") or tt:find("superhuman") or tt:find("buso")) then
                    hum:EquipTool(tool); return
                elseif wType == "Sword" and (tt:find("sword") or tt:find("saber") or tt:find("katana") or tt:find("trident") or tt:find("cutlass")) then
                    hum:EquipTool(tool); return
                elseif wType == "Gun" and (tt:find("gun") or tt:find("rifle") or tt:find("bazooka") or tt:find("musket") or tt:find("cannon")) then
                    hum:EquipTool(tool); return
                end
            end
        end
    end)
end

local function runMacro(macro)
    if isRunning then return end
    isRunning = true
    task.spawn(function()
        for _, step in ipairs(macro.steps) do
            if not isRunning then break end
            pcall(function()
                equipWeapon(step.weapon)
                task.wait(step.delay or 0)
                pressKey(step.skill, step.hold or 0.15)
            end)
        end
        isRunning = false
        _G.SacredMacroUpdateTrigger()
    end)
end

-- ── Helpers ───────────────────────────────────────────────────
local function corner(p, r) Instance.new("UICorner", p).CornerRadius = UDim.new(0, r or 8) end
local function stroke(p, col, thick, trans)
    local s = Instance.new("UIStroke", p)
    s.Color = col or C.border; s.Thickness = thick or 1; s.Transparency = trans or 0
    return s
end
local function mkFrame(parent, size, pos, bg, trans)
    local f = Instance.new("Frame", parent)
    f.Size = size; f.Position = pos or UDim2.new(0,0,0,0)
    f.BackgroundColor3 = bg or C.card; f.BackgroundTransparency = trans or 0
    f.BorderSizePixel = 0; corner(f); return f
end
local function mkLabel(parent, size, pos, text, font, ts, color, xAlign)
    local l = Instance.new("TextLabel", parent)
    l.Size = size; l.Position = pos or UDim2.new(0,0,0,0)
    l.BackgroundTransparency = 1; l.Text = text or ""
    l.Font = font or Enum.Font.GothamBold; l.TextSize = ts or 12
    l.TextColor3 = color or C.text
    l.TextXAlignment = xAlign or Enum.TextXAlignment.Center
    l.TextTruncate = Enum.TextTruncate.AtEnd; return l
end
local function mkBtn(parent, size, pos, text, bg, tc, ts)
    local b = Instance.new("TextButton", parent)
    b.Size = size; b.Position = pos or UDim2.new(0,0,0,0)
    b.BackgroundColor3 = bg or C.accent; b.BorderSizePixel = 0
    b.Text = text or ""; b.Font = Enum.Font.GothamBold
    b.TextSize = ts or 11; b.TextColor3 = tc or Color3.new(1,1,1)
    b.AutoButtonColor = false; corner(b); return b
end
local function mkScroll(parent, size, pos)
    local s = Instance.new("ScrollingFrame", parent)
    s.Size = size; s.Position = pos or UDim2.new(0,0,0,0)
    s.BackgroundTransparency = 1; s.BorderSizePixel = 0
    s.ScrollBarThickness = 3; s.ScrollBarImageColor3 = C.accent
    s.CanvasSize = UDim2.new(0,0,0,0); s.AutomaticCanvasSize = Enum.AutomaticSize.Y
    local layout = Instance.new("UIListLayout", s)
    layout.Padding = UDim.new(0,4); layout.SortOrder = Enum.SortOrder.LayoutOrder
    return s
end
local function mkTextBox(parent, size, pos, placeholder)
    local b = Instance.new("TextBox", parent)
    b.Size = size; b.Position = pos or UDim2.new(0,0,0,0)
    b.BackgroundColor3 = C.card; b.BorderSizePixel = 0
    b.PlaceholderText = placeholder or ""; b.Text = ""
    b.Font = Enum.Font.GothamBold; b.TextSize = 12
    b.TextColor3 = C.text; b.ClearTextOnFocus = false
    corner(b); stroke(b, C.border, 1, 0.3)
    local pad = Instance.new("UIPadding", b); pad.PaddingLeft = UDim.new(0,8)
    return b
end

    for _, ch in ipairs(skillRow:GetChildren()) do ch:Destroy() end

    local skills = WEAPONS[selectedWeapon] or {}
    local w = 1 / #skills

        local sb = mkBtn(skillRow, UDim2.new(w,-3,1,0),
            UDim2.new((i-1)*w, i > 1 and 3 or 0, 0, 0),
            sk, C.card, C.muted, 13)
        stroke(sb, C.border, 1, 0.5)
        skillBtns[sk] = sb
        sb.MouseButton1Click:Connect(function()
            selectedSkill = sk
            for k, b in pairs(skillBtns) do
                b.BackgroundColor3 = k == sk and C.accent or C.card
                b.TextColor3       = k == sk and Color3.fromRGB(8,10,20) or C.muted
            end
        end)
    end
    if not skillBtns[selectedSkill] then selectedSkill = skills[1] or "Z" end
    if skillBtns[selectedSkill] then
        skillBtns[selectedSkill].BackgroundColor3 = C.accent
        skillBtns[selectedSkill].TextColor3       = Color3.fromRGB(8,10,20)
    end
end

local function buildStepsUI()
    for _, ch in ipairs(stepsScroll:GetChildren()) do
        if ch:IsA("Frame") then ch:Destroy() end
    end
    if not currentEditMacro then return end
    for i, step in ipairs(currentEditMacro.steps) do
        local idx = i
        local row = mkFrame(stepsScroll, UDim2.new(1,-4,0,34), UDim2.new(0,2,0,0), C.card)
        row.LayoutOrder = i; stroke(row, C.border, 1, 0.55)

            step.skill, Enum.Font.GothamBlack, 14, C.accent)
        mkLabel(row, UDim2.new(0,100,0.5,0), UDim2.new(0,24,0.5,0),
            step.delay .. "s espera  " .. step.hold .. "s hold",
            Enum.Font.Gotham, 8, C.muted, Enum.TextXAlignment.Left)

        local editBtn = mkBtn(row, UDim2.new(0,26,0,22), UDim2.new(1,-58,0.5,-11),
            "e", Color3.fromRGB(18,40,70), C.accent, 11)

            editingStepIdx = idx
            local s = currentEditMacro.steps[idx]
            selectedWeapon = s.weapon; selectedSkill = s.skill
            delayBox.Text  = tostring(s.delay); holdBox.Text = tostring(s.hold)
            for k, b in pairs(weaponBtns) do
                b.BackgroundColor3 = k == selectedWeapon and C.purple or C.card
                b.TextColor3 = k == selectedWeapon and Color3.new(1,1,1) or C.muted
            end
            updateSkillButtons()
            if skillBtns[selectedSkill] then
                skillBtns[selectedSkill].BackgroundColor3 = C.accent
                skillBtns[selectedSkill].TextColor3 = Color3.fromRGB(8,10,20)
            end
            stepModal.Visible = true
        end)

        local delBtn = mkBtn(row, UDim2.new(0,26,0,22), UDim2.new(1,-28,0.5,-11),
            "x", Color3.fromRGB(65,15,15), C.red, 11)
        delBtn.MouseButton1Click:Connect(function()
            table.remove(currentEditMacro.steps, idx)
            buildStepsUI()
        end)
    end
end

    if editingStepIdx then
        currentEditMacro.steps[editingStepIdx] = step

    end

saveBtn.MouseButton1Click:Connect(function()
    if not currentEditMacro then return end
    local name = nameBox.Text ~= "" and nameBox.Text or ("Macro " .. (#macros+1))
    currentEditMacro.name = name
    if currentEditMacro._idx then
        macros[currentEditMacro._idx] = {name=name, steps=currentEditMacro.steps}
    else
        table.insert(macros, {name=name, steps=currentEditMacro.steps})
    end
    saveMacros()
    _G.SacredMacroRebuild()
    editorScreen.Visible = false
    emptyScreen.Visible  = true
end)

-- ══════════════════════════════════════════════════════════════
--  LISTA DE MACROS
-- ══════════════════════════════════════════════════════════════
local function rebuildMacroList()
    for _, ch in ipairs(macroScroll:GetChildren()) do
        if ch:IsA("Frame") then ch:Destroy() end
    end
    for i, macro in ipairs(macros) do
        local idx    = i
        local isAct  = activeMacro == i
        local row    = mkFrame(macroScroll, UDim2.new(1,-4,0,44), UDim2.new(0,2,0,0),
            isAct and Color3.fromRGB(12,18,34) or C.card)
        row.LayoutOrder = i
        stroke(row, isAct and C.accent or C.border, 1, isAct and 0.25 or 0.6)

        mkLabel(row, UDim2.new(1,-56,0.52,0), UDim2.new(0,8,0,2),
            macro.name, Enum.Font.GothamBold, 12, isAct and C.text or C.muted,
            Enum.TextXAlignment.Left)
        mkLabel(row, UDim2.new(1,-56,0.42,0), UDim2.new(0,8,0.54,0),
            #macro.steps .. " pasos", Enum.Font.Gotham, 9, C.muted, Enum.TextXAlignment.Left)

        -- Activar / Desactivar (toggle)
        local actBtn = mkBtn(row, UDim2.new(0,22,0,18), UDim2.new(1,-50,0.5,-9),
            isAct and "■" or "▶", isAct and C.red or C.accent, Color3.new(1,1,1), 10)
        actBtn.MouseButton1Click:Connect(function()
            if activeMacro == idx then
                -- Desactivar
                activeMacro = nil
                triggerBtn.Visible = false
                isRunning = false
            else
                -- Activar
                activeMacro = idx
                triggerBtn.Visible = true
            end
            _G.SacredMacroUpdateTrigger()
            rebuildMacroList()
        end)

        -- Editar
        local editBtn = mkBtn(row, UDim2.new(0,22,0,18), UDim2.new(1,-26,0.5,-9),
            "✎", C.card, C.muted, 11)
        editBtn.MouseButton1Click:Connect(function()
            currentEditMacro = {name=macro.name, steps={}, _idx=idx}
            for _, s in ipairs(macro.steps) do
                table.insert(currentEditMacro.steps, {weapon=s.weapon, skill=s.skill, delay=s.delay, hold=s.hold})
            end
            nameBox.Text = macro.name
            buildStepsUI()
            emptyScreen.Visible  = false
            editorScreen.Visible = true
            stepModal.Visible    = false
        end)

        -- Borrar
        local delBtn = mkBtn(row, UDim2.new(0,16,0,16), UDim2.new(1,-186,0.5,-8),
            "x", Color3.fromRGB(60,18,18), C.red, 9)
        delBtn.MouseButton1Click:Connect(function()
            table.remove(macros, idx)
            if activeMacro == idx then
                activeMacro = nil; triggerBtn.Visible = false
            elseif activeMacro and activeMacro > idx then
                activeMacro -= 1
            end
            saveMacros(); rebuildMacroList()
            emptyScreen.Visible  = true
            editorScreen.Visible = false
        end)
    end
end

_G.SacredMacroRebuild = rebuildMacroList

newMacroBtn.MouseButton1Click:Connect(function()
    currentEditMacro = {name="", steps={}, _idx=nil}
    nameBox.Text = ""; buildStepsUI()
    emptyScreen.Visible  = false
    editorScreen.Visible = true
    stepModal.Visible    = false
end)

rebuildMacroList()
